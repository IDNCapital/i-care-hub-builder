### 1. Creation de `src/components/ErrorBoundary.tsx`

    import { Component } from "react";
    import type { ErrorInfo, ReactNode } from "react";

    interface Props {
      children: ReactNode;
    }

    interface State {
      enErreur: boolean;
    }

    /**
     * Barriere d'erreur React. Composant de classe : les deux crochets
     * getDerivedStateFromError et componentDidCatch n'existent pas en
     * composant de fonction.
     *
     * Repli sobre, en francais, avec un bouton qui recharge la page.
     * Aucune telemetrie, aucun appel reseau, aucune dependance nouvelle.
     */
    export class ErrorBoundary extends Component<Props, State> {
      state: State = { enErreur: false };

      static getDerivedStateFromError(): State {
        return { enErreur: true };
      }

      componentDidCatch(error: Error, info: ErrorInfo): void {
        // Volontairement muet en production : la decision de jour du 01/09
        // exclut toute telemetrie. Les deux arguments sont conserves pour
        // que la signature du crochet reste celle que React appelle.
        void error;
        void info;
      }

      render() {
        if (this.state.enErreur) {
          return (
            <div role="alert" className="flex min-h-[50vh] flex-col items-center justify-center gap-4 px-6 text-center">
              <h1 className="text-xl font-semibold">Une erreur est survenue</h1>
              <p className="max-w-prose text-sm text-muted-foreground">
                Cette page n'a pas pu s'afficher. Vous pouvez recharger la page.
              </p>
              <button
                type="button"
                onClick={() => window.location.reload()}
                className="rounded-md border px-4 py-2 text-sm font-medium"
              >
                Recharger la page
              </button>
            </div>
          );
        }

        return this.props.children;
      }
    }

### Les deux crochets, cites comme la mission l'exige

`getDerivedStateFromError`, methode statique, rend le nouvel etat sans effet de bord : c'est elle qui bascule l'affichage vers le repli. `componentDidCatch`, methode d'instance, recoit l'erreur et la pile de composants : c'est l'endroit ou une telemetrie irait, et la decision de jour l'interdit, donc elle reste muette. Les deux sont necessaires et ne se remplacent pas l'une l'autre.
